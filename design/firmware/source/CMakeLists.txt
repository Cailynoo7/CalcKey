cmake_minimum_required(VERSION 3.20)
set(PICO_BOARD waveshare_rp2350_plus_16mb CACHE STRING "Target board")
if(NOT PICO_BOARD STREQUAL "waveshare_rp2350_plus_16mb")
    message(FATAL_ERROR "CalcKey hardware uses Waveshare RP2350-Plus-16MB; configure PICO_BOARD=waveshare_rp2350_plus_16mb")
endif()
if(NOT DEFINED PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
    set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
endif()
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(calckey VERSION 0.3.0 LANGUAGES C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
add_executable(calckey main.cpp decimal.cpp application.cpp lcd.cpp usb_descriptors.cpp)
target_include_directories(calckey PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_compile_options(calckey PRIVATE -Wall -Wextra -Werror -fstack-usage)
target_compile_definitions(calckey PRIVATE PICO_STACK_SIZE=4096)
target_link_libraries(calckey pico_stdlib pico_unique_id tinyusb_device)
pico_enable_stdio_usb(calckey 0)
pico_enable_stdio_uart(calckey 0)
pico_set_program_name(calckey "CalcKey")
pico_set_program_version(calckey "${PROJECT_VERSION}")
pico_add_extra_outputs(calckey)
